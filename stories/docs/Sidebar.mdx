# Sidebar

A minimal sidebar navigation component. Supports light and dark themes, collapsible icon-only mode, nested nav sections with badges, and pinned bottom actions. Can be imported as `meticulous-ui/components/Sidebar`.

## NavItem shape

```ts
{
  id: string;
  label: string;
  icon: ReactNode;
  badge?: number;           // numeric badge shown on the right
  onAdd?: () => void;       // renders a + button; called on click
  children?: {              // collapsible sub-items
    id: string;
    label: string;
    badge?: number;
  }[];
}
```

## BottomItem shape

```ts
{
  id: string;
  label?: string;   // shown in expanded mode
  icon: ReactNode;
  onClick?: () => void;
}
```

## Usage

```jsx
import Sidebar from 'meticulous-ui/components/Sidebar';

const NAV_ITEMS = [
  { id: 'analytics', label: 'Analytics', icon: <AnalyticsIcon /> },
  {
    id: 'posts',
    label: 'Posts',
    icon: <PostsIcon />,
    onAdd: handleAddPost,
    children: [
      { id: 'drafts', label: 'Drafts', badge: 10 },
      { id: 'published', label: 'Published', badge: 28 },
    ],
  },
];

const BOTTOM_ITEMS = [
  { id: 'settings', label: 'Settings', icon: <SettingsIcon />, onClick: openSettings },
];

function App() {
  const [active, setActive] = useState('analytics');
  const [collapsed, setCollapsed] = useState(false);

  return (
    <Sidebar
      logo={<Logo />}
      storeName='Acme Corp'
      storeUrl='acme.store.com'
      onSwitchStore={handleSwitch}
      navItems={NAV_ITEMS}
      activeId={active}
      onNavChange={setActive}
      bottomItems={BOTTOM_ITEMS}
      collapsed={collapsed}
      onCollapseToggle={() => setCollapsed((c) => !c)}
      theme='light'
    />
  );
}
```
